1//! Configuration module for [`MlsGroup`] configurations.
2//!
3//! ## Building an MlsGroupCreateConfig
4//! The [`MlsGroupCreateConfigBuilder`] makes it easy to build configurations for the
5//! [`MlsGroup`].
6//!
7//! ```
8//! use openmls::prelude::*;
9//!
10//! let group_config = MlsGroupCreateConfig::builder()
11//!     .use_ratchet_tree_extension(true)
12//!     .build();
13//! ```
14//!
15//! See [`MlsGroupCreateConfigBuilder`](MlsGroupCreateConfigBuilder#implementations) for
16//! all options that can be configured.
17//!
18//! ### Wire format policies
19//! Only some combination of possible wire formats are valid within OpenMLS.
20//! The [`WIRE_FORMAT_POLICIES`] lists all valid options that can be set.
21//!
22//! ```
23//! use openmls::prelude::*;
24//!
25//! let group_config = MlsGroupCreateConfig::builder()
26//!     .wire_format_policy(MIXED_CIPHERTEXT_WIRE_FORMAT_POLICY)
27//!     .build();
28//! ```

47/// Configures the automatic deletion of past epoch secrets.
48///
49/// **WARNING**
50///
51/// Policies other than `MaxEpochs(0)` enable the storage of message secrets from past epochs.
52/// It is a trade-off between functionality and forward secrecy and should only be enabled
53/// if the Delivery Service cannot guarantee that application messages will be sent in
54/// the same epoch in which they were generated. The number for `max_epochs` should be
55/// as low as possible.
56///
57/// If the MaxEpochs policy is set to MaxEpochs(usize::MAX), if will be returned
58/// as KeepAll after deserialization due to backwards compatibility constraints.
59#[derive(Clone, Debug, PartialEq, Eq)]
60pub enum PastEpochDeletionPolicy {
61    /// Keep at most `n` past epoch secrets.
62    MaxEpochs(usize),
63    /// Keep all past epoch secrets.
64    ///
65    /// NOTE: The application is responsible for deleting past epoch secrets when
66    /// `KeepAll` is set. Past epoch secrets can be deleted manually using:
67    /// - [`MlsGroup::delete_past_epoch_secrets()`]
68    KeepAll,
69}
70
71impl Default for PastEpochDeletionPolicy {
72    fn default() -> Self {
73        Self::MaxEpochs(0)
74    }
75}
76
77impl Serialize for PastEpochDeletionPolicy {
78    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
79    where
80        S: serde::Serializer,
81    {
82        let usize = match self {
83            Self::MaxEpochs(epochs) => *epochs,
84            Self::KeepAll => usize::MAX,
85        };
86        serializer.serialize_u64(usize as u64)
87    }
88}

90impl<'de> Deserialize<'de> for PastEpochDeletionPolicy {
91    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
92        // Previously supported format (changed to plain integer in 8bdba6f)
93        #[derive(Deserialize)]
94        enum Tagged {
95            MaxEpochs(usize),
96            KeepAll,
97        }
98
99        #[derive(Deserialize)]
100        #[serde(untagged)]
101        enum Format {
102            Int(u64),
103            Tagged(Tagged),
104        }
105
106        Ok(match Format::deserialize(deserializer)? {
107            Format::Int(u64::MAX) => Self::KeepAll,
108            Format::Int(n) => {
109                Self::MaxEpochs(usize::try_from(n).map_err(serde::de::Error::custom)?)
110            }
111            Format::Tagged(Tagged::MaxEpochs(n)) => Self::MaxEpochs(n),
112            Format::Tagged(Tagged::KeepAll) => Self::KeepAll,
113        })
114    }
115}

117/// The input to [`MlsGroup::delete_past_epoch_secrets()`].
118///
119/// This struct can be used for manual deletion of past epoch secrets by the application.
120///
121/// An [`MlsGroup`] also applies automatic deletion of past epoch secrets by default.
122///
123/// For more information, see [`PastEpochDeletionPolicy`] and [`MlsGroup::set_past_epoch_deletion_policy()`].
124///
125/// These methods can be used by the application to set up time-based deletion schedules:
126/// - [`PastEpochDeletion::before_timestamp()`]
127/// - [`PastEpochDeletion::older_than_duration()`]
128///
129/// **NOTE**: Epoch secrets that were created using `openmls=0.8.1` or earlier will not yet include a timestamp.
130/// After migration, these may not always be deleted by applying a time-based [`PastEpochDeletion`]. Only if a new secret that does include a timestamp is added later, and it matches the time-based condition in the [`PastEpochDeletion`], all earlier past epoch secrets without timestamps will be deleted, as well. However, otherwise, past epoch secrets without timestamps will not be affected by applying time-based [`PastEpochDeletion`]s.
131///
132/// To manually delete all past epoch secrets without timestamps, see:
133/// [`PastEpochDeletion::delete_all_without_timestamps()`]
134pub struct PastEpochDeletion {
135    pub(crate) config: Option<PastEpochDeletionTimeConfig>,
136    pub(crate) max_past_epochs: Option<usize>,
137}
138
139/// A duration or timestamp before which to delete past epoch secrets.
140pub(crate) enum PastEpochDeletionTimeConfig {
141    OlderThanDuration(std::time::Duration),
142    BeforeTimestamp(SystemTime),
143    DeleteAllWithoutTimestamp,
144}
145
146impl PastEpochDeletion {
147    /// Delete all past epoch secrets older than a provided duration.
148    pub fn older_than_duration(duration: std::time::Duration) -> Self {
149        Self {
150            config: Some(PastEpochDeletionTimeConfig::OlderThanDuration(duration)),
151            max_past_epochs: None,
152        }
153    }
154
155    /// Delete all past epoch secrets before a provided timestamp.
156    pub fn before_timestamp(timestamp: SystemTime) -> Self {
157        Self {
158            config: Some(PastEpochDeletionTimeConfig::BeforeTimestamp(timestamp)),
159            max_past_epochs: None,
160        }
161    }
162
163    /// Delete all past epoch secrets without timestamps.
164    ///
165    /// NOTE: This will delete all past epoch secrets having the legacy
166    /// format that does not include a timestamp.
167    pub fn delete_all_without_timestamps() -> Self {
168        Self {
169            config: Some(PastEpochDeletionTimeConfig::DeleteAllWithoutTimestamp),
170            max_past_epochs: None,
171        }
172    }
173
174    /// Delete all past epoch secrets.
175    pub fn delete_all() -> Self {
176        Self {
177            config: None,
178            max_past_epochs: None,
179        }
180    }
181
182    /// Set the number of `max_past_epochs` that should be kept, at most.
183    pub fn max_past_epochs(mut self, max_past_epochs: usize) -> Self {
184        self.max_past_epochs = Some(max_past_epochs);
185        self
186    }
187}

189impl PastEpochDeletionPolicy {
190    pub(crate) fn max_epochs(&self) -> Option<usize> {
191        match self {
192            Self::MaxEpochs(epochs) => Some(*epochs),
193            Self::KeepAll => None,
194        }
195    }

198/// Configures how many virtual-clients derivation epochs an emulation group
199/// keeps, so that delayed messages from sibling emulator clients can still be
200/// processed.
201///
202/// Registering a new derivation epoch drops the epochs beyond the window. Their
203/// key material is deleted unless a higher-level group or a retained KeyPackage
204/// still references it.
205#[cfg(feature = "virtual-clients-draft")]
206#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
207pub enum VcDerivationEpochRetentionPolicy {
208    /// Keep at most `n` derivation epochs.
209    MaxEpochs(usize),
210    /// Keep every derivation epoch. The application deletes them with
211    /// [`MlsGroup::delete_vc_derivation_epochs()`].
212    KeepAll,
213}
214
215#[cfg(feature = "virtual-clients-draft")]
216impl Default for VcDerivationEpochRetentionPolicy {
217    fn default() -> Self {
218        Self::MaxEpochs(5)
219    }
220}
221
222#[cfg(feature = "virtual-clients-draft")]
223impl VcDerivationEpochRetentionPolicy {
224    pub(crate) fn max_epochs(&self) -> Option<usize> {
225        match self {
226            Self::MaxEpochs(epochs) => Some(*epochs),
227            Self::KeepAll => None,
228        }
229    }
230}

377    /// Flag marking the created group as an emulation group of a virtual
378    /// client. Only consulted at group creation, the group keeps the flag

698    /// epoch that virtual-client operations resolve to. It may be older than

703    /// [`CommitBuilder::derivation_epoch`]: crate::group::CommitBuilder::derivation_epoch

766/// Defines what wire format is acceptable for incoming handshake messages.
767/// Note that application messages must always be encrypted.
768#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
769pub enum IncomingWireFormatPolicy {
770    /// Handshake messages must always be PrivateMessage
771    AlwaysCiphertext,
772    /// Handshake messages must always be PublicMessage
773    AlwaysPlaintext,
774    /// Handshake messages can either be PrivateMessage or PublicMessage
775    Mixed,
776}
777
778impl IncomingWireFormatPolicy {
779    pub(crate) fn is_compatible_with(&self, wire_format: WireFormat) -> bool {
780        match self {
781            IncomingWireFormatPolicy::AlwaysCiphertext => wire_format == WireFormat::PrivateMessage,
782            IncomingWireFormatPolicy::AlwaysPlaintext => wire_format == WireFormat::PublicMessage,
783            IncomingWireFormatPolicy::Mixed => {
784                wire_format == WireFormat::PrivateMessage
785                    || wire_format == WireFormat::PublicMessage
786            }
787        }
788    }
789}
790
791/// Defines what wire format should be used for outgoing handshake messages.
792/// Note that application messages must always be encrypted.
793#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
794pub enum OutgoingWireFormatPolicy {
795    /// Handshake messages must always be PrivateMessage
796    AlwaysCiphertext,
797    /// Handshake messages must always be PublicMessage
798    AlwaysPlaintext,
799}
800
801/// Defines what wire format is desired for outgoing handshake messages.
802/// Note that application messages must always be encrypted.
803#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
804pub struct WireFormatPolicy {
805    outgoing: OutgoingWireFormatPolicy,
806    incoming: IncomingWireFormatPolicy,
807}
808
809impl WireFormatPolicy {
810    /// Creates a new wire format policy from an [`OutgoingWireFormatPolicy`]
811    /// and an [`IncomingWireFormatPolicy`].
812    #[cfg(any(feature = "test-utils", test))]
813    pub(crate) fn new(
814        outgoing: OutgoingWireFormatPolicy,
815        incoming: IncomingWireFormatPolicy,
816    ) -> Self {
817        Self { outgoing, incoming }
818    }
819
820    /// Returns a reference to the wire format policy's outgoing wire format policy.
821    pub fn outgoing(&self) -> OutgoingWireFormatPolicy {
822        self.outgoing
823    }
824
825    /// Returns a reference to the wire format policy's incoming wire format policy.
826    pub fn incoming(&self) -> IncomingWireFormatPolicy {
827        self.incoming
828    }
829}
830
831impl Default for WireFormatPolicy {
832    fn default() -> Self {
833        PURE_CIPHERTEXT_WIRE_FORMAT_POLICY
834    }
835}
836
837impl From<OutgoingWireFormatPolicy> for WireFormat {
838    fn from(outgoing: OutgoingWireFormatPolicy) -> Self {
839        match outgoing {
840            OutgoingWireFormatPolicy::AlwaysCiphertext => WireFormat::PrivateMessage,
841            OutgoingWireFormatPolicy::AlwaysPlaintext => WireFormat::PublicMessage,
842        }
843    }
844}

888    #[test]
889    fn past_epoch_deletion_policy_roundtrip() {
890        for policy in [
891            PastEpochDeletionPolicy::MaxEpochs(0),
892            PastEpochDeletionPolicy::MaxEpochs(42),
893            PastEpochDeletionPolicy::KeepAll,
894        ] {
895            let json = serde_json::to_string(&policy).unwrap();
896            let deserialized: PastEpochDeletionPolicy = serde_json::from_str(&json).unwrap();
897            assert_eq!(deserialized, policy);
898        }
899
900        // MaxEpochs(usize::MAX) is indistinguishable from KeepAll after serialization.
901        let json = serde_json::to_string(&PastEpochDeletionPolicy::MaxEpochs(usize::MAX)).unwrap();
902        let deserialized: PastEpochDeletionPolicy = serde_json::from_str(&json).unwrap();
903        assert_eq!(deserialized, PastEpochDeletionPolicy::KeepAll);
904    }
905
906    #[test]
907    fn past_epoch_deletion_policy_deserializes_plain_integer() {
908        let deserialized: PastEpochDeletionPolicy = serde_json::from_str("42").unwrap();
909        assert_eq!(deserialized, PastEpochDeletionPolicy::MaxEpochs(42));
910
911        let deserialized: PastEpochDeletionPolicy =
912            serde_json::from_str(&u64::MAX.to_string()).unwrap();
913        assert_eq!(deserialized, PastEpochDeletionPolicy::KeepAll);
914    }
