1//! External Proposals

3//! Contains the types and methods to build external proposals of types
4//! `add``, `remove`, `psk``, `reinit``, and `group_context_extensions`.

25/// External Add Proposal where sender is [NewMemberProposal](crate::prelude::Sender::NewMemberProposal). A client
26/// outside the group can request joining the group. This proposal should then be committed by a
27/// group member. Note that this is unconstrained i.e. it works for any [MLS group](crate::group::MlsGroup).
28/// This is not the case for the same external proposal with a [Preconfigured sender](crate::prelude::Sender::External)
29pub struct JoinProposal;
30
31/// External Proposal where sender is [External](crate::prelude::Sender::External). A party
32/// outside the group can request to add or remove a member to the group. This proposal should then
33/// be committed by a group member. The sender must be pre configured within the group through the [crate::extensions::ExternalSendersExtension]
34pub struct ExternalProposal;
35
36impl JoinProposal {
37    /// Creates an external Add proposal. For clients requesting to be added to a group. This
38    /// proposal will have to be committed later by a group member.
39    ///
40    /// # Arguments
41    /// * `key_package` - of the joiner
42    /// * `group_id` - unique group identifier of the group to join
43    /// * `epoch` - group's epoch
44    /// * `signer` - of the sender to sign the message
45    #[allow(clippy::new_ret_no_self)]
46    pub fn new<Storage: StorageProvider>(
47        key_package: KeyPackage,
48        group_id: GroupId,
49        epoch: GroupEpoch,
50        signer: &impl Signer,
51    ) -> Result<MlsMessageOut, ProposeAddMemberError<Storage::Error>> {
52        AuthenticatedContent::new_join_proposal(
53            Proposal::add(AddProposal { key_package }),
54            group_id,
55            epoch,
56            signer,
57        )
58        .map(PublicMessage::from)
59        .map(MlsMessageOut::from)
60        .map_err(ProposeAddMemberError::from)
61    }
62}

64impl ExternalProposal {
65    /// Creates an external GroupContextExtensions proposal.
66    /// For example, for delivery services requesting to update the group context
67    /// extension.
68    /// This proposal will have to be committed later by a group member.
69    ///
70    /// # Arguments
71    /// * `extensions` - a new set of extensions for the group context
72    /// * `group_id` - unique group identifier of the group to join
73    /// * `epoch` - group's epoch
74    /// * `signer` - of the sender to sign the message
75    /// * `sender_index` - index of the sender of the proposal
76    ///   (in the [ExternalSendersExtension](crate::extensions::ExternalSendersExtension)
77    ///   array from the Group Context)
78    pub fn new_group_context_extensions<Provider: OpenMlsProvider>(
79        extensions: Extensions<GroupContext>,
80        group_id: GroupId,
81        epoch: GroupEpoch,
82        signer: &impl Signer,
83        sender_index: SenderExtensionIndex,
84    ) -> Result<MlsMessageOut, CreateGroupContextExtProposalError<Provider::StorageError>> {
85        let proposal = GroupContextExtensionProposal::new(extensions);
86
87        AuthenticatedContent::new_external_proposal(
88            Proposal::GroupContextExtensions(Box::new(proposal)),
89            group_id,
90            epoch,
91            signer,
92            sender_index,
93        )
94        .map(PublicMessage::from)
95        .map(MlsMessageOut::from)
96        .map_err(CreateGroupContextExtProposalError::from)
97    }

99    /// Creates an external Remove proposal.
100    /// For example, for delivery services requesting to remove a client.
101    /// This proposal will have to be committed later by a group member.
102    ///
103    /// # Arguments
104    /// * `removed` - index of the client to remove
105    /// * `group_id` - unique group identifier of the group to join
106    /// * `epoch` - group's epoch
107    /// * `signer` - of the sender to sign the message
108    /// * `sender_index` - index of the sender of the proposal
109    ///   (in the [ExternalSendersExtension](crate::extensions::ExternalSendersExtension)
110    ///   array from the Group Context)
111    pub fn new_remove<Provider: OpenMlsProvider>(
112        removed: LeafNodeIndex,
113        group_id: GroupId,
114        epoch: GroupEpoch,
115        signer: &impl Signer,
116        sender_index: SenderExtensionIndex,
117    ) -> Result<MlsMessageOut, ProposeRemoveMemberError<Provider::StorageError>> {
118        AuthenticatedContent::new_external_proposal(
119            Proposal::remove(RemoveProposal { removed }),
120            group_id,
121            epoch,
122            signer,
123            sender_index,
124        )
125        .map(PublicMessage::from)
126        .map(MlsMessageOut::from)
127        .map_err(ProposeRemoveMemberError::from)
128    }

134    /// # Arguments
135    /// * `key_package` - key package of the client to add
136    /// * `group_id` - unique group identifier of the group to join
137    /// * `epoch` - group's epoch
138    /// * `signer` - of the sender to sign the message
139    /// * `sender_index` - index of the sender of the proposal
140    ///   (in the [ExternalSendersExtension](crate::extensions::ExternalSendersExtension)
141    ///   array from the Group Context)
142    pub fn new_add<Provider: OpenMlsProvider>(
143        key_package: KeyPackage,
144        group_id: GroupId,
145        epoch: GroupEpoch,
146        signer: &impl Signer,
147        sender_index: SenderExtensionIndex,
148    ) -> Result<MlsMessageOut, ProposeAddMemberError<Provider::StorageError>> {
149        AuthenticatedContent::new_external_proposal(

159    }

161    /// Creates an external PreSharedKey proposal.
162    /// For example for delivery services requesting to inject a pre-shared key
163    /// into the group's key schedule (RFC 9420 §12.1.8 permits external senders
164    /// to send PSK proposals).
165    /// This proposal will have to be committed later by a group member.
166    /// The same constructor is used for both external and resumption PSKs;
167    /// the distinction is carried by the [`PreSharedKeyId`].
168    ///
169    /// # Arguments
170    /// * `psk_id` - identifier of the pre-shared key to inject
171    /// * `group_id` - unique group identifier of the group
172    /// * `epoch` - group's epoch
173    /// * `signer` - of the sender to sign the message
174    /// * `sender_index` - index of the sender of the proposal
175    ///   (in the [ExternalSendersExtension](crate::extensions::ExternalSendersExtension)
176    ///   array from the Group Context)
177    pub fn new_pre_shared_key(
178        psk_id: PreSharedKeyId,
179        group_id: GroupId,
180        epoch: GroupEpoch,
181        signer: &impl Signer,
182        sender_index: SenderExtensionIndex,
183    ) -> Result<MlsMessageOut, LibraryError> {
184        AuthenticatedContent::new_external_proposal(
185            Proposal::psk(PreSharedKeyProposal::new(psk_id)),
186            group_id,
187            epoch,
188            signer,
189            sender_index,
190        )
191        .map(PublicMessage::from)
192        .map(MlsMessageOut::from)
193    }
194
195    /// Creates an external ReInit proposal. For delivery services requesting to
196    /// reinitialize the group as a new group with new parameters (RFC 9420
197    /// §12.1.8.2 permits external senders to send ReInit proposals). This
198    /// proposal will have to be committed later by a group member, which
199    /// suspends the old group; a member then creates the successor group from
200    /// the ReInit parameters.
201    ///
202    /// # Arguments
203    /// * `reinit` - the successor group's parameters (its own new group id,
204    ///   protocol version, ciphersuite, and group context extensions)
205    /// * `group_id` - unique group identifier of the *old* group the proposal
206    ///   is sent into
207    /// * `epoch` - old group's epoch
208    /// * `signer` - of the sender to sign the message
209    /// * `sender_index` - index of the sender of the proposal (in the
210    ///   [crate::extensions::ExternalSendersExtension] array from the Group Context)
211    pub fn new_reinit(
212        reinit: ReInitProposal,
213        group_id: GroupId,
214        epoch: GroupEpoch,
215        signer: &impl Signer,
216        sender_index: SenderExtensionIndex,
217    ) -> Result<MlsMessageOut, LibraryError> {
218        AuthenticatedContent::new_external_proposal(
219            Proposal::re_init(reinit),
220            group_id,
221            epoch,
222            signer,
223            sender_index,
224        )
225        .map(PublicMessage::from)
226        .map(MlsMessageOut::from)
227    }
