60impl tls_codec::Size for FrankenPublicMessage {
61    fn tls_serialized_len(&self) -> usize {
62        let tag_len = self
63            .membership_tag
64            .as_ref()
65            .map_or(0, |tag| tag.tls_serialized_len());
66
67        self.content.tls_serialized_len() + self.auth.tls_serialized_len() + tag_len
68    }
69}

71impl Deserialize for FrankenPublicMessage {
72    fn tls_deserialize<R: std::io::prelude::Read>(bytes: &mut R) -> Result<Self, Error>
73    where
74        Self: Sized,
75    {
76        let content = FrankenFramedContent::tls_deserialize(bytes)?;
77        let auth = if matches!(content.body, FrankenFramedContentBody::Commit(_)) {
78            FrankenFramedContentAuthData::tls_deserialize_with_tag(bytes)?
79        } else {
80            FrankenFramedContentAuthData::tls_deserialize_without_tag(bytes)?
81        };
82
83        let membership_tag = if matches!(content.sender, FrankenSender::Member(_)) {
84            Some(VLBytes::tls_deserialize(bytes)?)
85        } else {
86            None
87        };
88
89        Ok(Self {
90            content,
91            auth,
92            membership_tag,
93        })
94    }
95}

97impl DeserializeBytes for FrankenPublicMessage {
98    fn tls_deserialize_bytes(bytes: &[u8]) -> Result<(Self, &[u8]), Error>
99    where
100        Self: Sized,
101    {
102        let (content, bytes) = FrankenFramedContent::tls_deserialize_bytes(bytes)?;
103        let (auth, bytes) = match content.body {
104            FrankenFramedContentBody::Commit(_) => {
105                FrankenFramedContentAuthData::tls_deserialize_bytes_with_tag(bytes)
106            }
107            _ => FrankenFramedContentAuthData::tls_deserialize_bytes_without_tag(bytes),
108        }?;
109        let (membership_tag, bytes) = match content.sender {
110            FrankenSender::Member(_) => {
111                let (tag, bytes) = VLBytes::tls_deserialize_bytes(bytes)?;
112                (Some(tag), bytes)
113            }
114            _ => (None, bytes),
115        };
116
117        Ok((
118            Self {
119                content,
120                auth,
121                membership_tag,
122            },
123            bytes,
124        ))
125    }
126}

128impl Serialize for FrankenPublicMessage {

130        let mut written = 0;

132        written += self.auth.tls_serialize(writer)?;

141impl FrankenPublicMessage {
142    /// auth builds a mostly(!) valid fake public message. However, it does not compute a correct
143    /// confirmation_tag. If the caller wants to process a message that requires a
144    /// confirmation_tag, they have two options:
145    ///
146    /// 1. build a valid tag themselves and provide it through the option
147    /// 2. provide a dummy tag and disable the verification of confirmation tags using
148    ///    [`crate::disable_confirmation_tag_verification`].
149    ///    NB: Usually, confirmation tag verification should be turned back on after the call that
150    ///    needs to be tricked!
151    pub(crate) fn auth(
152        provider: &impl crate::storage::OpenMlsProvider,
153        ciphersuite: openmls_traits::types::Ciphersuite,
154        signer: &impl Signer,
155        content: FrankenFramedContent,
156        group_context: Option<&FrankenGroupContext>,
157        membership_key: Option<&[u8]>,
158        confirmation_tag: Option<VLBytes>,
159    ) -> Self {
160        let version = 1; // MLS 1.0
161        let wire_format = 1; // PublicMessage
162
163        let franken_tbs = FrankenFramedContentTbs {
164            version,
165            wire_format,
166            content: &content,
167            group_context,
168        };
169
170        let auth = FrankenFramedContentAuthData::build(
171            signer,
172            version,
173            wire_format,
174            &content,
175            group_context,
176            confirmation_tag,
177        );
178
179        let tbm = FrankenAuthenticatedContentTbm {
180            content_tbs: franken_tbs,
181            auth: auth.clone(),
182        };
183
184        let membership_tag = membership_key.map(|membership_key| {
185            compute_membership_tag(provider.crypto(), ciphersuite, membership_key, &tbm)
186        });
187
188        FrankenPublicMessage {
189            content,
190            auth,
191            membership_tag,
192        }
193    }
194}

311impl tls_codec::Size for FrankenFramedContentAuthData {

323        let mut written = 0;
324        written += self.signature.tls_serialize(writer)?;

332impl FrankenFramedContentAuthData {
333    pub fn build(
334        signer: &impl Signer,
335        version: u16,
336        wire_format: u16,
337        content: &FrankenFramedContent,
338        group_context: Option<&FrankenGroupContext>,
339        confirmation_tag: Option<VLBytes>,
340    ) -> Self {
341        let content_tbs = FrankenFramedContentTbs {
342            version,
343            wire_format,
344            content,
345            group_context,
346        };
347
348        let content_tbs_serialized = content_tbs.tls_serialize_detached().unwrap();
349
350        let signature =
351            sign_with_label(signer, b"FramedContentTBS", &content_tbs_serialized).into();
352
353        Self {
354            signature,
355            confirmation_tag,
356        }
357    }
358}

360#[derive(Debug, Clone, PartialEq, Eq)]
361pub struct FrankenFramedContentTbs<'a> {
362    version: u16,
363    wire_format: u16,
364    content: &'a FrankenFramedContent,
365    group_context: Option<&'a FrankenGroupContext>,
366}
367
368impl tls_codec::Size for FrankenFramedContentTbs<'_> {
369    fn tls_serialized_len(&self) -> usize {
370        if let Some(ctx) = self.group_context {
371            4 + self.content.tls_serialized_len() + ctx.tls_serialized_len()
372        } else {
373            4 + self.content.tls_serialized_len()
374        }
375    }
376}
377
378impl Serialize for FrankenFramedContentTbs<'_> {
379    fn tls_serialize<W: std::io::Write>(&self, writer: &mut W) -> Result<usize, Error> {
380        writer.write_all(&self.version.to_be_bytes())?;
381        writer.write_all(&self.wire_format.to_be_bytes())?;
382
383        let mut written = 4; // contains the two u16 version and wire_format
384        written += self.content.tls_serialize(writer)?;
385        if let Some(group_context) = &self.group_context {
386            written += group_context.tls_serialize(writer)?;
387        }
388
389        Ok(written)
390    }
391}

527impl From<Sender> for FrankenSender {
528    fn from(value: Sender) -> Self {
529        match value {
530            Sender::Member(i) => FrankenSender::Member(i.u32()),
531            // this cast is safe, because the index method casts it from u32 to usize for some
532            // reason, so it's known to fit u32
533            Sender::External(i) => FrankenSender::External(i.index() as u32),
534            Sender::NewMemberProposal => FrankenSender::NewMemberProposal,
535            Sender::NewMemberCommit => FrankenSender::NewMemberCommit,
536        }
537    }
538}
