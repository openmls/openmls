21/// How a leaf's capabilities are treated when they don't already cover what
22/// the leaf being built needs.
23///
24/// There is deliberately no [`Default`]: which policy applies depends on
25/// whether the caller set capabilities at all, so every call site resolves it
26/// through `resolve_capabilities` rather than falling back on its own.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
28pub enum CapabilitiesPolicy {
29    /// Reject construction unless the given capabilities already cover the
30    /// leaf's own ciphersuite, credential type and extension types, and the
31    /// group's required capabilities.
32    Reject,
33    /// Add whatever the leaf itself uses — its ciphersuite, credential type
34    /// and extension types — and reject only if the group's required
35    /// capabilities are still unmet. Never removes anything the caller set.
36    ///
37    /// Note that this widens only capabilities the library can derive from the
38    /// leaf. It never claims support for the group's required extension,
39    /// proposal or credential types on its own, and it cannot
40    /// cover the cross-member credential checks a group applies on receipt.
41    /// Therefore, a widened leaf is not automatically an acceptable one.
42    Widen,
43}
44
45/// Resolve builder-level capabilities and policy into what leaf construction
46/// needs.
47///
48/// Capabilities the caller never set are widened from the leaf itself: there is
49/// exactly one sensible answer for what an unconfigured leaf should advertise,
50/// and it is derivable. Capabilities the caller set explicitly are held to
51/// exactly what they listed, because a list that doesn't cover the leaf is a
52/// mistake worth reporting rather than papering over.
53///
54/// An explicit policy always wins over both defaults.
55pub(crate) fn resolve_capabilities(
56    capabilities: Option<Capabilities>,
57    policy: Option<CapabilitiesPolicy>,
58) -> (Capabilities, CapabilitiesPolicy) {
59    match capabilities {
60        Some(capabilities) => (capabilities, policy.unwrap_or(CapabilitiesPolicy::Reject)),
61        None => (
62            Capabilities::default(),
63            policy.unwrap_or(CapabilitiesPolicy::Widen),
64        ),
65    }
66}
67
68/// Like [`resolve_capabilities`], but for a leaf that already exists in a tree.
69///
70/// Unset capabilities are inherited from that leaf rather than derived from
71/// scratch: they were already validated when it entered the tree, so there is
72/// nothing to widen from, and a rejection means the group's requirements have
73/// changed since.
74pub(crate) fn resolve_capabilities_for_existing_leaf(
75    capabilities: Option<Capabilities>,
76    policy: Option<CapabilitiesPolicy>,
77    existing: &Capabilities,
78) -> (Capabilities, CapabilitiesPolicy) {
79    match capabilities {
80        Some(_) => resolve_capabilities(capabilities, policy),
81        None => (
82            existing.clone(),
83            policy.unwrap_or(CapabilitiesPolicy::Reject),
84        ),
85    }
86}

121    ///
122    /// Only `versions` has a default (`Mls1.0`); every other `None`
123    /// produces an **empty** list. An empty list is an
124    /// explicit statement that nothing is supported, so a leaf built from it is
125    /// rejected unless the caller also asks for [`CapabilitiesPolicy::Widen`].
126    ///
127    /// Prefer [`Capabilities::builder`], which makes the empty starting point
128    /// obvious.
129    // TODO(#1232)
130    pub fn new(
131        versions: Option<&[ProtocolVersion]>,
132        ciphersuites: Option<&[Ciphersuite]>,
133        extensions: Option<&[ExtensionType]>,
134        proposals: Option<&[ProposalType]>,
135        credentials: Option<&[CredentialType]>,
136    ) -> Self {
137        Self {
138            versions: match versions {
139                Some(v) => v.into(),
140                None => default_versions(),

144                None => vec![],
145            },
146            extensions: match extensions {
147                Some(e) => e.into(),
148                None => vec![],
149            },
150            proposals: match proposals {
151                Some(p) => p.into(),
152                None => vec![],
153            },
154            credentials: match credentials {
155                Some(c) => c.into(),
156                None => vec![],
157            },
158        }
159    }

227    /// Check if these [`Capabilities`] support all the capabilities required by
228    /// the given [`RequiredCapabilitiesExtension`].
229    ///
230    /// # Errors
231    ///
232    /// Returns a [`LeafNodeValidationError`] error if any of the required
233    /// capabilities is not supported.
234    pub(crate) fn supports_required_capabilities(
235        &self,
236        required_capabilities: &RequiredCapabilitiesExtension,
237    ) -> Result<(), LeafNodeValidationError> {
238        // The required capabilities come from the wire, so all three checks use
239        // a set lookup and stop at the first unsupported entry.
240
241        // Check if all required extensions are supported.
242        let supported_extensions: HashSet<ExtensionType> =
243            self.extensions().iter().copied().collect();
244        if required_capabilities
245            .extension_types()
246            .iter()
247            .any(|e| !e.is_default() && !supported_extensions.contains(e))
248        {
249            log::error!(
250                "Leaf node does not support all required extension types\n
251                Supported extensions: {:?}\n
252                Required extensions: {:?}",
253                self.extensions(),
254                required_capabilities.extension_types()
255            );
256            return Err(LeafNodeValidationError::UnsupportedExtensions);
257        }
258        // Check if all required proposals are supported.
259        let supported_proposals: HashSet<ProposalType> = self.proposals().iter().copied().collect();
260        if required_capabilities
261            .proposal_types()
262            .iter()
263            .any(|p| !p.is_default() && !supported_proposals.contains(p))
264        {
265            return Err(LeafNodeValidationError::UnsupportedProposals);
266        }
267        // Check if all required credential types are supported.
268        let supported_credentials: HashSet<CredentialType> =
269            self.credentials().iter().copied().collect();
270        if required_capabilities
271            .credential_types()
272            .iter()
273            .any(|c| !supported_credentials.contains(c))
274        {
275            return Err(LeafNodeValidationError::UnsupportedCredentials);
276        }
277        Ok(())
278    }

280    /// Check if these [`Capabilities`] contain all the extensions.
281    pub(crate) fn contains_extensions(
282        &self,
283        extensions: &Extensions<impl ExtensionValidator>,
284    ) -> bool {
285        let mut required = extensions
286            .iter()
287            .map(Extension::extension_type)
288            .filter(|e| !e.is_default())
289            .peekable();
290
291        // Most leaf nodes carry no non-default extensions. Skip building the
292        // lookup set for them.
293        if required.peek().is_none() {
294            return true;
295        }
296
297        let supported: HashSet<ExtensionType> = self.extensions().iter().copied().collect();
298        required.all(|e| supported.contains(&e))
299    }

308        }

337            self.versions.push(version);
338        }

341    /// Widen `self` to cover what the leaf itself uses: its ciphersuite, its
342    /// credential type, and its non-default extension types.
343    ///
344    /// Only ever adds; never removes anything the caller set. Every added
345    /// entry is a fact about the leaf being built, so advertising it is
346    /// always truthful.
347    ///
348    /// Note what this deliberately does *not* do: it never adds the group's
349    /// [`RequiredCapabilitiesExtension`] entries. Those would be claims about
350    /// what the *application* implements, which the library is in no position
351    /// to make on its behalf, so a leaf that doesn't meet the group's
352    /// requirements is rejected instead — see
353    /// [`Capabilities::supports_required_capabilities`].
354    ///
355    /// Default extension types are skipped: RFC 9420 makes support for
356    /// them implicit, so they never need listing. GREASE extension types are
357    /// never default, so a GREASE extension actually present on the leaf is
358    /// covered here.
359    pub(super) fn widen_for(
360        &mut self,
361        ciphersuite: Ciphersuite,
362        credential_type: CredentialType,
363        leaf_extensions: &Extensions<LeafNode>,
364    ) {
365        let verifiable_ciphersuite = VerifiableCiphersuite::from(ciphersuite);
366        if !self.contains_ciphersuite(verifiable_ciphersuite) {
367            self.ciphersuites.push(verifiable_ciphersuite);
368        }
369
370        if !self.contains_credential(credential_type) {
371            self.credentials.push(credential_type);
372        }
373
374        for extension_type in leaf_extensions.iter().map(Extension::extension_type) {
375            if !extension_type.is_default() && !self.extensions.contains(&extension_type) {
376                self.extensions.push(extension_type);
377            }
378        }
379    }

381    /// Add random GREASE values to the capabilities to ensure extensibility.
382    ///
383    /// This adds one random GREASE value to each capability list if no GREASE
384    /// value is already present:
385    /// - Ciphersuites
386    /// - Extensions
387    /// - Proposals
388    /// - Credentials
389    ///
390    /// GREASE values are used per [RFC 9420 Section 13.5](https://www.rfc-editor.org/rfc/rfc9420.html#section-13.5)
391    /// to help prevent extensibility failures by ensuring implementations properly
392    /// handle unknown values.
393    ///
394    /// # Example
395    ///
396    /// ```
397    /// use openmls::prelude::*;
398    /// use openmls_rust_crypto::OpenMlsRustCrypto;
399    ///
400    /// let provider = OpenMlsRustCrypto::default();
401    ///
402    /// // Create capabilities with GREASE values injected
403    /// let capabilities = Capabilities::builder()
404    ///     .build()
405    ///     .with_grease(provider.rand());
406    ///
407    /// // Verify GREASE values were added
408    /// assert!(capabilities.ciphersuites().iter().any(|cs| cs.is_grease()));
409    /// assert!(capabilities.extensions().iter().any(|ext| ext.is_grease()));
410    /// assert!(capabilities.proposals().iter().any(|prop| prop.is_grease()));
411    /// assert!(capabilities.credentials().iter().any(|cred| cred.is_grease()));
412    /// ```
413    pub fn with_grease(mut self, rand: &impl openmls_traits::random::OpenMlsRand) -> Self {
414        use crate::credentials::CredentialType;
415        use crate::extensions::ExtensionType;
416        use crate::messages::proposals::ProposalType;
417        use openmls_traits::types::VerifiableCiphersuite;
418
419        // Add GREASE ciphersuite if none present
420        if !self.ciphersuites.iter().any(|cs| cs.is_grease()) {
421            let grease_cs = VerifiableCiphersuite::new(crate::grease::random_grease_value(rand));
422            self.ciphersuites.push(grease_cs);
423        }
424
425        // Add GREASE extension if none present
426        if !self.extensions.iter().any(|ext| ext.is_grease()) {
427            let grease_ext = ExtensionType::Grease(crate::grease::random_grease_value(rand));
428            self.extensions.push(grease_ext);
429        }
430
431        // Add GREASE proposal if none present
432        if !self.proposals.iter().any(|prop| prop.is_grease()) {
433            let grease_prop = ProposalType::Grease(crate::grease::random_grease_value(rand));
434            self.proposals.push(grease_prop);
435        }
436
437        // Add GREASE credential if none present
438        if !self.credentials.iter().any(|cred| cred.is_grease()) {
439            let grease_cred = CredentialType::Grease(crate::grease::random_grease_value(rand));
440            self.credentials.push(grease_cred);
441        }
442
443        self
444    }
